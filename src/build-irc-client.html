protocol walkthrough

Write your own IRC client

IRC is plain text over a TCP socket, which makes it one of the easiest network protocols to implement. Below: the handful of commands you need, and the same small client in nine languages.

The protocol

Every message is a single line terminated by \r\n (carriage return + line feed). A minimal client needs six commands:

NICK <nickname>
Set your nickname
USER <username> 0 * :<realname>
Identify yourself to the server
JOIN <#channel>
Join a channel
PRIVMSG <target> :<message>
Send a message to a channel or user
PING <token> / PONG <token>
Keep-alive: when the server sends PING, answer with PONG and the same token
QUIT :<message>
Disconnect from the server

Open a socket to port 6667 (6697 for TLS), send NICK and USER to register, answer PINGs to stay connected, then JOIN channels and PRIVMSG to talk. A short session looks like this, with > for what the client sends and < for what the server sends back:

Each example below does exactly that: connects to Libera Chat, joins #test, says hello and quits, answering PINGs along the way.

Python

Only the standard library: socket plus ssl for TLS. Tested with Python 3.6 and 3.14.

import socket
import ssl
import sys

server = "irc.libera.chat"
port = 6697
nick = "mybot"
channel = "#test"

sock = socket.create_connection((server, port))
irc = ssl.create_default_context().wrap_socket(sock, server_hostname=server)

def send(msg):
    irc.sendall(f"{msg}\r\n".encode())

send(f"NICK {nick}")
send(f"USER {nick} 0 * :{nick}")

buf = b""
while True:
    data = irc.recv(4096)
    if not data:  # the server closed the connection
        break
    buf += data
    while b"\r\n" in buf:
        line, buf = buf.split(b"\r\n", 1)
        line = line.decode("utf-8", errors="replace")
        print(line)

        if line.startswith("PING"):
            send("PONG" + line[4:])

        # 001 = RPL_WELCOME, meaning registration is complete
        if " 001 " in line:
            send(f"JOIN {channel}")
            send(f"PRIVMSG {channel} :Hello from Python!")
            send("QUIT :Bye")
            irc.close()
            sys.exit()

irc.close()

JavaScript (Node.js)

The built-in tls module. Tested with Node.js 8 and 26.

const tls = require("tls");

const server = "irc.libera.chat";
const port = 6697;
const nick = "mybot";
const channel = "#test";

const socket = tls.connect(port, server, () => {
  send(`NICK ${nick}`);
  send(`USER ${nick} 0 * :${nick}`);
});

function send(msg) {
  socket.write(msg + "\r\n");
}

let buffer = "";
socket.setEncoding("utf8");
socket.on("data", (data) => {
  buffer += data;
  const lines = buffer.split("\r\n");
  buffer = lines.pop();

  for (const line of lines) {
    console.log(line);

    if (line.startsWith("PING")) {
      send("PONG" + line.slice(4));
    }

    // 001 = RPL_WELCOME
    if (line.includes(" 001 ")) {
      send(`JOIN ${channel}`);
      send(`PRIVMSG ${channel} :Hello from Node.js!`);
      send("QUIT :Bye");
    }
  }
});

Go

Standard library only: crypto/tls and bufio. Tested with Go 1.13 and 1.27.

package main

import (
	"bufio"
	"crypto/tls"
	"fmt"
	"strings"
)

func main() {
	server := "irc.libera.chat:6697"
	nick := "mybot"
	channel := "#test"

	conn, err := tls.Dial("tcp", server, nil)
	if err != nil {
		panic(err)
	}
	defer conn.Close()

	send := func(msg string) {
		fmt.Fprintf(conn, "%s\r\n", msg)
	}

	send("NICK " + nick)
	send("USER " + nick + " 0 * :" + nick)

	scanner := bufio.NewScanner(conn)
	for scanner.Scan() {
		line := scanner.Text()
		fmt.Println(line)

		if strings.HasPrefix(line, "PING") {
			send("PONG" + line[4:])
		}

		// 001 = RPL_WELCOME
		if strings.Contains(line, " 001 ") {
			send("JOIN " + channel)
			send("PRIVMSG " + channel + " :Hello from Go!")
			send("QUIT :Bye")
			return
		}
	}
}

Rust

The standard library plus the native-tls crate for TLS. Needs Rust 1.80 or newer, because the current TLS crates do. Tested with 1.80 and 1.98.

use native_tls::TlsConnector;
use std::io::{BufRead, BufReader, Write};
use std::net::TcpStream;

fn send(stream: &mut impl Write, msg: &str) {
    write!(stream, "{msg}\r\n").unwrap();
    stream.flush().unwrap();
}

fn main() {
    let server = "irc.libera.chat";
    let port = 6697;
    let nick = "mybot";
    let channel = "#test";

    let connector = TlsConnector::new().unwrap();
    let tcp = TcpStream::connect(format!("{server}:{port}")).unwrap();
    let tls = connector.connect(server, tcp).unwrap();

    let mut reader = BufReader::new(tls);

    send(reader.get_mut(), &format!("NICK {nick}"));
    send(reader.get_mut(), &format!("USER {nick} 0 * :{nick}"));

    let mut line = String::new();
    while reader.read_line(&mut line).unwrap() > 0 {
        print!("{line}");

        if line.starts_with("PING") {
            let pong = format!("PONG {}", line[4..].trim());
            send(reader.get_mut(), &pong);
        }

        // 001 = RPL_WELCOME
        if line.contains(" 001 ") {
            send(reader.get_mut(), &format!("JOIN {channel}"));
            send(reader.get_mut(), &format!("PRIVMSG {channel} :Hello from Rust!"));
            send(reader.get_mut(), "QUIT :Bye");
            break;
        }

        line.clear();
    }
}

C

POSIX sockets over plain TCP. For anything real you would add TLS with OpenSSL. Tested with GCC 7 and 16.

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <netdb.h>
#include <sys/socket.h>

static int irc_fd;

void irc_send(const char *msg) {
    dprintf(irc_fd, "%s\r\n", msg);
}

/* Strip trailing \r\n */
void chomp(char *s) {
    size_t len = strlen(s);
    while (len > 0 && (s[len - 1] == '\r' || s[len - 1] == '\n'))
        s[--len] = '\0';
}

int main(void) {
    const char *server = "irc.libera.chat";
    const char *port = "6667";
    const char *nick = "mybot";
    const char *channel = "#test";

    struct addrinfo hints = { .ai_socktype = SOCK_STREAM }, *res, *ai;
    int err = getaddrinfo(server, port, &hints, &res);
    if (err != 0) {
        fprintf(stderr, "getaddrinfo: %s\n", gai_strerror(err));
        return 1;
    }

    /* Try each address (IPv6, IPv4) until one connects */
    irc_fd = -1;
    for (ai = res; ai != NULL; ai = ai->ai_next) {
        irc_fd = socket(ai->ai_family, ai->ai_socktype, ai->ai_protocol);
        if (irc_fd == -1)
            continue;
        if (connect(irc_fd, ai->ai_addr, ai->ai_addrlen) == 0)
            break;
        close(irc_fd);
        irc_fd = -1;
    }
    freeaddrinfo(res);
    if (irc_fd == -1) {
        fprintf(stderr, "could not connect to %s\n", server);
        return 1;
    }

    char buf[512];
    snprintf(buf, sizeof(buf), "NICK %s", nick);
    irc_send(buf);
    snprintf(buf, sizeof(buf), "USER %s 0 * :%s", nick, nick);
    irc_send(buf);

    char line[512];
    FILE *fp = fdopen(dup(irc_fd), "r");
    while (fgets(line, sizeof(line), fp)) {
        chomp(line);
        printf("%s\n", line);

        if (strncmp(line, "PING", 4) == 0) {
            snprintf(buf, sizeof(buf), "PONG%s", line + 4);
            irc_send(buf);
        }

        /* 001 = RPL_WELCOME */
        if (strstr(line, " 001 ")) {
            snprintf(buf, sizeof(buf), "JOIN %s", channel);
            irc_send(buf);
            snprintf(buf, sizeof(buf), "PRIVMSG %s :Hello from C!", channel);
            irc_send(buf);
            irc_send("QUIT :Bye");
            break;
        }
    }

    fclose(fp);
    close(irc_fd);
    return 0;
}

/* Compile: gcc -o irc irc.c */

Java

SSLSocketFactory gives you TLS out of the box. Tested with Java 8 and 25.

import javax.net.ssl.SSLSocketFactory;
import java.io.*;
import java.net.Socket;
import java.nio.charset.StandardCharsets;

public class IrcClient {
    static OutputStream raw;

    static void send(String msg) throws Exception {
        raw.write((msg + "\r\n").getBytes(StandardCharsets.UTF_8));
        raw.flush();
    }

    public static void main(String[] args) throws Exception {
        String server = "irc.libera.chat";
        int port = 6697;
        String nick = "mybot";
        String channel = "#test";

        Socket socket = SSLSocketFactory.getDefault()
            .createSocket(server, port);
        raw = socket.getOutputStream();
        BufferedReader in = new BufferedReader(
            new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));

        send("NICK " + nick);
        send("USER " + nick + " 0 * :" + nick);

        String line;
        while ((line = in.readLine()) != null) {
            System.out.println(line);

            if (line.startsWith("PING")) {
                send("PONG" + line.substring(4));
            }

            // 001 = RPL_WELCOME
            if (line.contains(" 001 ")) {
                send("JOIN " + channel);
                send("PRIVMSG " + channel + " :Hello from Java!");
                send("QUIT :Bye");
                break;
            }
        }

        socket.close();
    }
}

Ruby

The openssl standard library handles TLS. Tested with Ruby 2.3 and 4.0.

require "socket"
require "openssl"

server = "irc.libera.chat"
port = 6697
nick = "mybot"
channel = "#test"

tcp = TCPSocket.new(server, port)
ctx = OpenSSL::SSL::SSLContext.new
ctx.set_params(verify_mode: OpenSSL::SSL::VERIFY_PEER)
irc = OpenSSL::SSL::SSLSocket.new(tcp, ctx)
irc.sync_close = true
irc.hostname = server
irc.connect
irc.post_connection_check(server)

def irc_send(conn, msg)
  conn.write("#{msg}\r\n")
end

irc_send(irc, "NICK #{nick}")
irc_send(irc, "USER #{nick} 0 * :#{nick}")

while line = irc.gets
  line.chomp!
  puts line

  if line.start_with?("PING")
    irc_send(irc, "PONG#{line[4..-1]}")
  end

  # 001 = RPL_WELCOME
  if line.include?(" 001 ")
    irc_send(irc, "JOIN #{channel}")
    irc_send(irc, "PRIVMSG #{channel} :Hello from Ruby!")
    irc_send(irc, "QUIT :Bye")
    break
  end
end

C#

SslStream from .NET. Uses top-level statements, so it needs .NET 6 or newer. Tested with .NET 6 and 10.

using System.Net.Security;
using System.Net.Sockets;
using System.Text;

var server = "irc.libera.chat";
var port = 6697;
var nick = "mybot";
var channel = "#test";

using var tcp = new TcpClient(server, port);
using var ssl = new SslStream(tcp.GetStream());
ssl.AuthenticateAsClient(server);

using var reader = new StreamReader(ssl, Encoding.UTF8);

void Send(string msg)
{
    var bytes = Encoding.UTF8.GetBytes(msg + "\r\n");
    ssl.Write(bytes);
    ssl.Flush();
}

Send($"NICK {nick}");
Send($"USER {nick} 0 * :{nick}");

string? line;
while ((line = reader.ReadLine()) != null)
{
    Console.WriteLine(line);

    if (line.StartsWith("PING"))
        Send("PONG" + line[4..]);

    // 001 = RPL_WELCOME
    if (line.Contains(" 001 "))
    {
        Send($"JOIN {channel}");
        Send($"PRIVMSG {channel} :Hello from C#!");
        Send("QUIT :Bye");
        break;
    }
}

PHP

The ssl:// stream wrapper opens a TLS socket. Tested with PHP 7.0 and 8.5.

<?php

$server = "irc.libera.chat";
$port = 6697;
$nick = "mybot";
$channel = "#test";

$irc = stream_socket_client("ssl://{$server}:{$port}");

function irc_send($conn, $msg) {
    fwrite($conn, "{$msg}\r\n");
}

irc_send($irc, "NICK {$nick}");
irc_send($irc, "USER {$nick} 0 * :{$nick}");

while ($line = fgets($irc)) {
    $line = rtrim($line, "\r\n");
    echo $line . "\n";

    if (strncmp($line, "PING", 4) === 0) {
        irc_send($irc, "PONG" . substr($line, 4));
    }

    // 001 = RPL_WELCOME
    if (strpos($line, " 001 ") !== false) {
        irc_send($irc, "JOIN {$channel}");
        irc_send($irc, "PRIVMSG {$channel} :Hello from PHP!");
        irc_send($irc, "QUIT :Bye");
        break;
    }
}

fclose($irc);

What a real client needs

These examples are the bare minimum. A client people would actually use also has to deal with:

Message parsing
The full message format, :prefix COMMAND params :trailing, plus IRCv3 tags
Multiple channels
Tracking which channels you are in and routing each message to the right one
Nick collision
433 ERR_NICKNAMEINUSE means your nick is taken, so try another
Reconnection
Connections drop. Reconnect and rejoin without the user noticing
SASL authentication
Logging in to a registered account using IRCv3 SASL
CAP negotiation
Asking the server for IRCv3 capabilities such as message history and tags
User interface
A terminal UI, a desktop window or a web page. This is where most of the work goes
